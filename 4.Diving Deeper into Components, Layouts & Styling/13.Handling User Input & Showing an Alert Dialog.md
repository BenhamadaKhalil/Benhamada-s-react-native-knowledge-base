
```tsx
import React, { useState } from 'react';
import { View, TextInput, Alert, StyleSheet } from 'react-native';
import PrimaryButton from '../components/PrimaryButton';

export default function StartGameScreen() {
  // State to store the user's input as a string
  const [enteredNumber, setEnteredNumber] = useState('');

  // Handler for text input
  function numberInputHandler(inputText: string) {
    setEnteredNumber(inputText);
  }

  // Reset the input to empty string
  function resetInputHandler() {
    setEnteredNumber('');
  }

  // Confirm input handler with validation
  function confirmInputHandler() {
    const chosenNumber = parseInt(enteredNumber); // Convert string to number

    // Validate input: must be a number between 1 and 99
    if (isNaN(chosenNumber) || chosenNumber <= 0 || chosenNumber > 99) {
      Alert.alert(
        'Invalid Number',
        'Number has to be a number between 1 and 99.',
        [
          {
            text: 'Okay',
            style: 'destructive',
            onPress: resetInputHandler, // Clear input after alert
          },
        ]
      );
      return; // Stop execution if invalid
    }

    // Valid number: do something with it (currently logging)
    console.log('Valid number chosen:', chosenNumber);
    // TODO: Navigate to game screen here
  }

  return (
    <View style={styles.inputContainer}>
      <TextInput
        style={styles.numberInput}
        keyboardType="number-pad"
        maxLength={2}
        onChangeText={numberInputHandler}
        value={enteredNumber}
      />
      <View style={styles.buttonsContainer}>
        <PrimaryButton onPress={resetInputHandler}>Reset</PrimaryButton>
        <PrimaryButton onPress={confirmInputHandler}>Confirm</PrimaryButton>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  inputContainer: {
    marginTop: 100,
    alignItems: 'center',
    padding: 16,
    backgroundColor: '#f0e5d8',
    borderRadius: 8,
  },
  numberInput: {
    height: 50,
    width: 50,
    fontSize: 32,
    borderBottomColor: '#72063c',
    borderBottomWidth: 2,
    marginVertical: 16,
    textAlign: 'center',
  },
  buttonsContainer: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    width: '60%',
  },
});
```

---

### ✅ What this does:

1. **State Handling:** `enteredNumber` keeps track of the text input.
    
2. **Input Conversion:** `parseInt` converts it to a number for validation.
    
3. **Validation Checks:**
    
    - Not a number → alert.
        
    - Number ≤ 0 or > 99 → alert.
        
4. **Alert Handling:** Shows a platform-native alert using `Alert.alert`.
    
5. **Reset Input:** Input is cleared when alert is confirmed or when the Reset button is pressed.
    
6. **Buttons:** Uses your reusable `PrimaryButton` component.
